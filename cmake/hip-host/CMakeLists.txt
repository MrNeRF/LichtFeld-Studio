cmake_minimum_required(VERSION 3.30)
project(LichtFeldHipHost LANGUAGES CXX)
enable_testing()
get_filename_component(_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(_hip "${_root}/src/core/tensor/backend/hip")
add_library(hip_host_module STATIC "${_hip}/runtime_module.cpp" "${_hip}/runtime_context.cpp")
target_include_directories(hip_host_module PUBLIC "${_hip}")
target_compile_features(hip_host_module PUBLIC cxx_std_20)
target_link_libraries(hip_host_module PRIVATE ${CMAKE_DL_LIBS})
add_executable(hip_host_module_tests "${_root}/tests/hip/test_tensor_hip_runtime_module.cpp")
target_link_libraries(hip_host_module_tests PRIVATE hip_host_module)
foreach(_revision 1 2 3 4 5 6)
    add_library(hip_module_fixture_${_revision} SHARED "${_root}/tests/hip/module_fixture.cpp")
    target_compile_definitions(hip_module_fixture_${_revision} PRIVATE HIP_FIXTURE_ABI=${_revision})
    add_test(NAME HipHost.RejectRevision${_revision}
        COMMAND hip_host_module_tests "$<TARGET_FILE:hip_module_fixture_${_revision}>" ${_revision})
    set_tests_properties(HipHost.RejectRevision${_revision} PROPERTIES LABELS "cpu;hip" TIMEOUT 30)
endforeach()
set(LFS_HIP_HOST_TEST_MODULE "" CACHE FILEPATH "Optional real HIP module for CPU-only export checks")
set(LFS_HIP_HOST_TEST_RUNTIME_DIR "" CACHE PATH "Optional explicit SDK dependency directory for Windows tests")
if(LFS_HIP_HOST_TEST_MODULE)
    add_test(NAME HipHost.NativeExports COMMAND hip_host_module_tests "${LFS_HIP_HOST_TEST_MODULE}" 0 "${LFS_HIP_HOST_TEST_RUNTIME_DIR}")
    set_tests_properties(HipHost.NativeExports PROPERTIES LABELS "cpu;hip" TIMEOUT 30)
endif()

option(LFS_HIP_HOST_GPU_TESTS "Run explicit host adapter GPU context tests" OFF)
if(LFS_HIP_HOST_GPU_TESTS)
    if(NOT LFS_HIP_HOST_TEST_MODULE)
        message(FATAL_ERROR "Host GPU tests require an explicit native module")
    endif()
    find_package(Threads REQUIRED)
    add_executable(hip_host_context_tests "${_root}/tests/hip/test_tensor_hip_runtime_context.cpp")
    target_link_libraries(hip_host_context_tests PRIVATE hip_host_module Threads::Threads)
    add_test(NAME HipHost.ContextOwnership COMMAND hip_host_context_tests
        "${LFS_HIP_HOST_TEST_MODULE}" "${LFS_HIP_HOST_TEST_RUNTIME_DIR}")
    set_tests_properties(HipHost.ContextOwnership PROPERTIES LABELS "gpu;hip" TIMEOUT 60)
    set(LFS_HIP_HOST_FAULT_MODULE "" CACHE FILEPATH "Optional instrumented module for explicit GPU recovery tests")
    if(LFS_HIP_HOST_FAULT_MODULE)
        add_test(NAME HipHost.ContextRecovery COMMAND hip_host_context_tests
            "${LFS_HIP_HOST_FAULT_MODULE}" "${LFS_HIP_HOST_TEST_RUNTIME_DIR}" faults)
        set_tests_properties(HipHost.ContextRecovery PROPERTIES LABELS "gpu;hip" TIMEOUT 60)
    endif()
endif()
