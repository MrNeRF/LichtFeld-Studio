# Standalone native protocol tests: no CUDA, desktop or full app build needed.
cmake_minimum_required(VERSION 3.25)
project(LichtFeldNativeDragContracts LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Use the application BUILD_SPLAT_TRANSFER_TESTS target on Windows")
endif()
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(NATIVE_DRAG REQUIRED IMPORTED_TARGET gtk+-3.0 x11 wayland-client wayland-server)
find_path(LFS_SDL_INCLUDE_DIR SDL3/SDL.h REQUIRED)
set(PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(BACKEND "${PROJECT_ROOT}/src/visualizer/gui/utils")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/resources/assets")
configure_file("${PROJECT_ROOT}/src/visualizer/gui/assets/lichtfeld-icon.png" "${CMAKE_BINARY_DIR}/resources/assets/lichtfeld-icon.png" COPYONLY)
option(LFS_DRAG_SANITIZERS "Enable address, undefined-behavior and leak checks" OFF)
add_executable(wayland_contract wayland_contract.cpp "${BACKEND}/native_file_drag.cpp" "${BACKEND}/native_file_drag_wayland.cpp")
target_compile_definitions(wayland_contract PRIVATE LFS_HAS_WAYLAND_DRAG=1)
add_executable(x11_guard_contract native_linux_contract.cpp "${BACKEND}/native_file_drag.cpp" "${BACKEND}/native_file_drag_wayland.cpp")
add_executable(progress_contract progress_contract.cpp)
foreach(target IN ITEMS wayland_contract x11_guard_contract progress_contract)
    target_include_directories(${target} PRIVATE "${PROJECT_ROOT}/src" "${PROJECT_ROOT}/src/visualizer" "${PROJECT_ROOT}/src/core/include" "${LFS_SDL_INCLUDE_DIR}")
    target_compile_features(${target} PRIVATE cxx_std_23)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(${target} PRIVATE PkgConfig::NATIVE_DRAG Threads::Threads)
    if(LFS_DRAG_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
# The mock compositor implements only requests used by the protocol contract;
# unused server request slots are intentionally zero. No production suppression.
set_source_files_properties(wayland_contract.cpp PROPERTIES COMPILE_OPTIONS "-Wno-missing-field-initializers")
enable_testing()
add_test(NAME wayland_drag_contract COMMAND wayland_contract)
add_test(NAME x11_drag_guard_contract COMMAND x11_guard_contract)
add_test(NAME file_transfer_progress_contract COMMAND progress_contract)
set_tests_properties(wayland_drag_contract x11_drag_guard_contract file_transfer_progress_contract PROPERTIES TIMEOUT 30)
