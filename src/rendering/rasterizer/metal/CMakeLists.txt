if(NOT APPLE)
    return()
endif()
enable_language(OBJCXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/splat_preprocessor.metal" LFS_METAL_PREPROCESS_SOURCE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/viewer_overlay.metal" LFS_METAL_OVERLAY_SOURCE)
set(_viewer_reference_config "${CMAKE_CURRENT_SOURCE_DIR}/../vulkan/shader/src/slang/config.slang")
file(READ "${_viewer_reference_config}" _viewer_reference_source)
foreach(_constant IN ITEMS HIGS_TILE_WIDTH HIGS_TILE_HEIGHT HIGS_MACRO_TILE_WIDTH_TILES HIGS_MACRO_TILE_HEIGHT_TILES)
    string(REGEX MATCH "#define ${_constant} ([0-9]+)u" _match "${_viewer_reference_source}")
    if(NOT _match)
        message(FATAL_ERROR "Missing Vulkan overlay reference constant: ${_constant}")
    endif()
    set(_${_constant} "${CMAKE_MATCH_1}")
endforeach()
math(EXPR _overlay_macro_width "${_HIGS_TILE_WIDTH} * ${_HIGS_MACRO_TILE_WIDTH_TILES}")
math(EXPR _overlay_macro_height "${_HIGS_TILE_HEIGHT} * ${_HIGS_MACRO_TILE_HEIGHT_TILES}")
string(REPLACE "@LFS_METAL_MACRO_LAYOUT@"
    "constant float2 overlay_macro_extent=float2(${_overlay_macro_width},${_overlay_macro_height});\nconstant float2 overlay_tile_extent=float2(${_HIGS_TILE_WIDTH},${_HIGS_TILE_HEIGHT});"
    LFS_METAL_OVERLAY_SOURCE "${LFS_METAL_OVERLAY_SOURCE}")
string(REGEX MATCH "#define NEAR_CLIP ([0-9.]+)f" _near_clip_match "${_viewer_reference_source}")
if(NOT _near_clip_match)
    message(FATAL_ERROR "Missing Vulkan viewer near-clip contract")
endif()
set(_viewer_near_clip "${CMAKE_MATCH_1}f")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_viewer_reference_config}")
string(REPLACE "@LFS_METAL_OVERLAY@" "${LFS_METAL_OVERLAY_SOURCE}" LFS_METAL_PREPROCESS_SOURCE "${LFS_METAL_PREPROCESS_SOURCE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/viewer_overlay.metal")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/splat_preprocessor.metal")
set(_portal_compact "${CMAKE_CURRENT_SOURCE_DIR}/../vulkan/shader/src/slang/portal_compact.inc")
set(_display_color "${CMAKE_CURRENT_SOURCE_DIR}/../vulkan/shader/src/slang/display_color.inc")
get_filename_component(_portal_compact "${_portal_compact}" ABSOLUTE)
get_filename_component(_display_color "${_display_color}" ABSOLUTE)
file(READ "${_portal_compact}" LFS_METAL_PORTAL_SOURCE)
file(READ "${_display_color}" LFS_METAL_DISPLAY_COLOR)
string(REPLACE "@LFS_METAL_PORTAL@" "${LFS_METAL_PORTAL_SOURCE}" LFS_METAL_PREPROCESS_SOURCE "${LFS_METAL_PREPROCESS_SOURCE}")
string(REPLACE "@LFS_METAL_DISPLAY_COLOR@" "${LFS_METAL_DISPLAY_COLOR}" LFS_METAL_PREPROCESS_SOURCE "${LFS_METAL_PREPROCESS_SOURCE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_portal_compact}" "${_display_color}")
configure_file(shader_source.hpp.in generated/shader_source.hpp @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/tile_rasterizer.metal" LFS_METAL_TILE_SOURCE)
string(REPLACE "@LFS_METAL_OVERLAY@" "${LFS_METAL_OVERLAY_SOURCE}" LFS_METAL_TILE_SOURCE "${LFS_METAL_TILE_SOURCE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/tile_rasterizer.metal")
configure_file(tile_shader_source.hpp.in generated/tile_shader_source.hpp @ONLY)
add_library(lfs_metal_viewer STATIC splat_preprocessor.mm tile_rasterizer.mm)
target_include_directories(lfs_metal_viewer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
    PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../core/include")
target_compile_features(lfs_metal_viewer PUBLIC cxx_std_20)
set_target_properties(lfs_metal_viewer PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
target_compile_options(lfs_metal_viewer PRIVATE -fobjc-arc)
target_compile_definitions(lfs_metal_viewer PRIVATE LFS_HAS_CUDA=0 PUBLIC LFS_METAL_VIEWER_NEAR_CLIP=${_viewer_near_clip})
target_link_libraries(lfs_metal_viewer PUBLIC "-framework Metal" "-framework Foundation")
