cmake_minimum_required(VERSION 3.24)
project(SceneUpscalerLoaderContracts LANGUAGES CXX)
find_package(GTest CONFIG REQUIRED)
find_package(Vulkan REQUIRED)
get_filename_component(HOST "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(scene_upscaler_loader_tests test_loader.cpp test_support.cpp
    "${HOST}/src/visualizer/rendering/scene_upscaler_plugin.cpp"
    "${HOST}/src/visualizer/rendering/scene_upscaler_registry.cpp")
target_compile_features(scene_upscaler_loader_tests PRIVATE cxx_std_23)
target_compile_definitions(scene_upscaler_loader_tests PRIVATE LFS_STATIC_BUILD LFS_GRAPHICS_VULKAN NOMINMAX)
target_include_directories(scene_upscaler_loader_tests PRIVATE "${HOST}/src/core/include" "${HOST}/src/visualizer")
target_link_libraries(scene_upscaler_loader_tests PRIVATE GTest::gtest Vulkan::Headers ${CMAKE_DL_LIBS})
if(WIN32)
    add_custom_command(TARGET scene_upscaler_loader_tests POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:GTest::gtest>" "$<TARGET_FILE_DIR:scene_upscaler_loader_tests>")
endif()

function(provider_fixture folder identifier name abi scale)
    add_library(${folder} SHARED fixture_plugin.cpp)
    target_compile_features(${folder} PRIVATE cxx_std_23)
    target_compile_definitions(${folder} PRIVATE LFS_SCENE_UPSCALER_PLUGIN_BUILD=1 "TEST_PROVIDER_ID=\"${identifier}\"" "TEST_PROVIDER_NAME=\"${name}\"" "TEST_PROVIDER_ABI=${abi}")
    target_include_directories(${folder} PRIVATE "${HOST}/src/visualizer")
    target_link_libraries(${folder} PRIVATE Vulkan::Headers)
    set_target_properties(${folder} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<TARGET_FILE_DIR:scene_upscaler_loader_tests>/scene_upscalers/${folder}" LIBRARY_OUTPUT_DIRECTORY "$<TARGET_FILE_DIR:scene_upscaler_loader_tests>/scene_upscalers/${folder}")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${folder}-metadata")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${folder}-metadata/provider-id.txt" "${identifier}\n")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${folder}-metadata/provider-module.txt" "${folder}\n")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${folder}-metadata/provider-presets.txt" "0.6666666865348816 ${scale} 0.5\n")
    add_custom_command(TARGET ${folder} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_BINARY_DIR}/${folder}-metadata" "$<TARGET_FILE_DIR:${folder}>")
endfunction()
provider_fixture(alpha provider-alpha "Provider Alpha" 1 0.58)
provider_fixture(beta amd-fsr3 "AMD FSR 3.1" 1 0.5882352590560913)
provider_fixture(bad bad-provider "Wrong ABI" 99 0.58)
provider_fixture(duplicate_a duplicate-provider "Duplicate A" 1 0.58)
provider_fixture(duplicate_b duplicate-provider "Duplicate B" 1 0.58)
enable_testing()
add_test(NAME generic_provider_loader COMMAND scene_upscaler_loader_tests --gtest_color=no)
