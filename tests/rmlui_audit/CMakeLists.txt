cmake_minimum_required(VERSION 3.30)
project(LichtFeldRmlUiAudit LANGUAGES CXX)
find_package(Freetype REQUIRED)
find_package(lunasvg CONFIG REQUIRED)
find_package(RmlUi CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(rmlui_layout_audit layout_audit.cpp)
target_compile_features(rmlui_layout_audit PRIVATE cxx_std_20)
target_link_libraries(rmlui_layout_audit PRIVATE RmlUi::RmlUi nlohmann_json::nlohmann_json)
option(LFS_RMLUI_AUDIT_USE_APP_CONTROLS "Use the 6.3 LichtFeld control dimension adapters" OFF)
if(LFS_RMLUI_AUDIT_USE_APP_CONTROLS)
    target_compile_definitions(rmlui_layout_audit PRIVATE LFS_RMLUI_AUDIT_USE_APP_CONTROLS)
    target_include_directories(rmlui_layout_audit PRIVATE ../../src/visualizer)
    find_package(GTest CONFIG REQUIRED)
    enable_testing()
    add_executable(rmlui_control_tests ../test_rml_form_control_layout.cpp)
    target_compile_features(rmlui_control_tests PRIVATE cxx_std_20)
    target_compile_definitions(rmlui_control_tests PRIVATE PROJECT_ROOT_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../..")
    target_include_directories(rmlui_control_tests PRIVATE ../../src/visualizer)
    target_link_libraries(rmlui_control_tests PRIVATE RmlUi::RmlUi GTest::gtest_main)
    add_test(NAME rmlui.controls COMMAND rmlui_control_tests)
endif()
