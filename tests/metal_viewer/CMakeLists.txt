cmake_minimum_required(VERSION 3.25)
project(LichtFeldMetalViewerContracts LANGUAGES CXX OBJCXX)
if(NOT APPLE)
    message(FATAL_ERROR "Metal viewer GPU contracts require an Apple host")
endif()
enable_testing()
add_subdirectory(../../src/rendering/rasterizer/metal metal_viewer)
add_executable(metal_viewer_contracts projection_contracts.mm)
target_link_libraries(metal_viewer_contracts PRIVATE lfs_metal_viewer)
target_compile_options(metal_viewer_contracts PRIVATE -fobjc-arc)
set_target_properties(metal_viewer_contracts PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
add_test(NAME MetalViewerProjectionContracts COMMAND metal_viewer_contracts)
set_tests_properties(MetalViewerProjectionContracts PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
add_executable(metal_raster_contracts raster_contracts.mm)
target_link_libraries(metal_raster_contracts PRIVATE lfs_metal_viewer)
target_compile_options(metal_raster_contracts PRIVATE -fobjc-arc)
set_target_properties(metal_raster_contracts PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
add_test(NAME MetalViewerRasterContracts COMMAND metal_raster_contracts)
set_tests_properties(MetalViewerRasterContracts PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
add_executable(viewer_backend_contracts backend_contracts.cpp)
target_compile_features(viewer_backend_contracts PRIVATE cxx_std_20)
target_include_directories(viewer_backend_contracts PRIVATE ../../src/rendering/include)
add_test(NAME ViewerBackendSelectionContracts COMMAND viewer_backend_contracts)
option(LFS_TEST_METAL_VULKAN_INTEROP "Test the installed desktop Vulkan ICD" OFF)
if(LFS_TEST_METAL_VULKAN_INTEROP)
    find_package(Vulkan REQUIRED)
    add_executable(metal_interop_contracts interop_contracts.mm)
    target_link_libraries(metal_interop_contracts PRIVATE Vulkan::Vulkan "-framework Metal" "-framework Foundation")
    target_compile_options(metal_interop_contracts PRIVATE -fobjc-arc)
    set_target_properties(metal_interop_contracts PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
    add_test(NAME MetalViewerInteropContracts COMMAND metal_interop_contracts)
    set_tests_properties(MetalViewerInteropContracts PROPERTIES TIMEOUT 30)
endif()
